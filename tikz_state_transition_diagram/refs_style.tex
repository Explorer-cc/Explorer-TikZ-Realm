\tikzset{
  base box/.style={%
    draw=black,
    fill=white,
    minimum width=2.2cm,
    minimum height=0.9cm,
    align=center,
  },
  terminal node/.style={%
    ellipse,
    base box,
    minimum width=1.8cm,
    minimum height=0.9cm,
    font=\ttfamily,
    inner sep=0pt
  },
  asm box/.style={%
    base box
  },
  diamond node/.style={%
    diamond,
    base box,
    aspect=2.444,
    inner sep=0pt
  },
  conditional box/.style={
    trapezium,base box,
    trapezium stretches=true,
    shape aspect=2.444,
    trapezium left angle=70,
    trapezium right angle=110,
    minimum height=0.9cm,
    inner sep=0pt
  },
  asm state/.style 2 args={%
    label={[font=\ttfamily, inner sep=1.5pt, anchor=south east]north east:#2},
    label={[font=, inner sep=1.5pt, anchor=south west]north west:#1}
  },
  asm dstate/.style 2 args={
    label={[font=\ttfamily, inner sep=1.5pt, anchor=center, xshift=0.3cm, yshift=0.3cm]north east:#2},
    label={[font=, inner sep=1.5pt, anchor=center, xshift=-0.3cm, yshift=0.3cm]north west:#1}
  },
  branch font/.style={
    font=\sffamily,
    fill=white,
    inner sep=1pt
  }
}
