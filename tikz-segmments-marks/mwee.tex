\ExplSyntaxOn
\keys_define:nn { cis / marksegment }
  {
    offset      .dim_set:N = \l_cis_offset_dim,
    offset     .initial:n = 10pt,
    bar-length  .dim_set:N = \l_cis_bar_length_dim,
    bar-length .initial:n = 5pt,
    bar-style   .tl_set:N  = \l_cis_bar_style_tl,
    bar-style  .initial:n = {thin},
    arrow       .tl_set:N  = \l_cis_arrow_tl,
    arrow      .initial:n = {Stealth-Stealth},
    arrow-style .tl_set:N  = \l_cis_arrow_style_tl,
    arrow-style.initial:n = {thin},
    node-style  .tl_set:N  = \l_cis_node_style_tl,
    node-style .initial:n = {fill=white,inner~sep=1.5pt},
  }

\cs_new_protected:Npn \cis_mark_segment:nnnn #1#2#3#4
  {
    \group_begin:
    \keys_set:nn { cis / marksegment } {#1}
    \use:e
      {
        \exp_not:N \path
          coordinate~(cis@from)~at~($~(#2)!\dim_use:N \l_cis_offset_dim!90\c_colon_str(#3)~$)
          coordinate~(cis@to)~at~($~(#3)!\dim_use:N \l_cis_offset_dim!-90\c_colon_str(#2)~$);
        \exp_not:N \draw [
          \exp_not:V \l_cis_arrow_tl,
          \exp_not:V \l_cis_arrow_style_tl
        ]
          (cis@from) -- (cis@to)
          node [midway, \exp_not:V \l_cis_node_style_tl] {\exp_not:n {#4}};
      }
    \dim_compare:nNnF \l_cis_bar_length_dim = \c_zero_dim
      {
        \use:e
          {
            \exp_not:N \draw [\exp_not:V \l_cis_bar_style_tl]
              ($~(cis@from)!\dim_eval:n { \l_cis_bar_length_dim / 2 }!90\c_colon_str(cis@to)~$) --
              ($~(cis@from)!-\dim_eval:n { \l_cis_bar_length_dim / 2 }!90\c_colon_str(cis@to)~$);
            \exp_not:N \draw [\exp_not:V \l_cis_bar_style_tl]
              ($~(cis@to)!\dim_eval:n { \l_cis_bar_length_dim / 2 }!-90\c_colon_str(cis@from)~$) --
              ($~(cis@to)!-\dim_eval:n { \l_cis_bar_length_dim / 2 }!-90\c_colon_str(cis@from)~$);
          }
      }
    \group_end:
  }

\NewDocumentCommand \marksegment { O{} m m m }
  {
    \cis_mark_segment:nnnn {#1} {#2} {#3} {#4}
  }
\ExplSyntaxOff
