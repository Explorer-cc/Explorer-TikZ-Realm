\def\arrowshift{0}

\newdimen\oplusarc@dim

\newcommand{\oplusarc}{%
  \mathbin{%
    \vphantom{\origoplus}%
    \mkern1.2mu
    \xmathpalette\newoplusarc@\origoplus
    \mkern1.2mu
  }%
}
\newcommand{\newoplusarc@}[3]{%
  \begingroup
  \sbox\z@{$#1\m@th#3$}%
  \oplusarc@dim=\dimeval{\ht\z@+\dp\z@}%
  \unitlength=0.065\oplusarc@dim%
  \def\pIIe@CAW{0.1\oplusarc@dim}%
  \smash{\vcenter{\hbox{%
    \begin{picture}(7,18)(0,-9)%
    \def\arrowshift{2}%
    \linethickness{\ifx#1\scriptscriptstyle 0.8\else0.7\fi\fontdimen8#2 3}%
    \VectorArc(0,0)(45:7){260}%{270} %
    \Line(-3,0)(3,0)%
    \Line(0,-3)(0,3)%
    \end{picture}%
  }}}%
  \endgroup
}
