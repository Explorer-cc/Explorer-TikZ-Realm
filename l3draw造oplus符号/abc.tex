\documentclass{article}
\usepackage{amsmath,pict2e}

\makeatletter
% Configuration constants
\newdimen\expl@cplus@totalheight    % Total height of reference + symbol
\newdimen\expl@cplus@symbolwidth    % Width of reference + symbol
\def\expl@cplus@thickness{0.06}     % Line thickness as fraction of height
\def\expl@cplus@circlesize{0.5}     % Circle diameter as fraction of height

% Main user command
% Usage: $a \explcplus b$
\newcommand{\explcplus}{%
  \mathbin{\mathpalette\expl@cplus@impl\relax}%
}

% Internal implementation
% #1 = math style (\displaystyle, \textstyle, etc.)
% #2 = unused (required by \mathpalette)
\newcommand{\expl@cplus@impl}[2]{%
  \vcenter{\hbox{%
    \expl@cplus@measure{#1}%
    \expl@cplus@draw
  }}%
}

% Measure reference + symbol in current math style
% Sets: \expl@cplus@totalheight, \expl@cplus@symbolwidth
\newcommand{\expl@cplus@measure}[1]{%
  \sbox\z@{$\m@th#1+$}%
  \expl@cplus@totalheight=\dimexpr\ht\z@ + \dp\z@\relax
  \expl@cplus@symbolwidth=\wd\z@
}

% Draw the circled plus using picture environment
\newcommand{\expl@cplus@draw}{%
  \linethickness{\expl@cplus@thickness\expl@cplus@totalheight}%
  \unitlength=\expl@cplus@totalheight
  %
  \begin{picture}(\expl@cplus@symbolwidth,1)
    % Circle centered at (width/2, 0.5)
    \put(0.5\expl@cplus@symbolwidth, 0.5){%
      \circle{\expl@cplus@circlesize}%
    }%
    % Vertical stroke from bottom to top
    \put(0.5\expl@cplus@symbolwidth, 0){%
      \line(0,1){1}%
    }%
    % Horizontal stroke from left to right
    \put(0, 0.5){%
      \line(1,0){\expl@cplus@symbolwidth}%
    }%
  \end{picture}%
}
\makeatother
