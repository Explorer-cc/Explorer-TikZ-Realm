\tikzset{seqpath/start/.initial={0,0},start/.style={seqpath/start={#1}}}
\ExplSyntaxOn
\msg_new:nnn {seqpath} {step} {Unknown~step~`#1';~use~H,~U,~or~D.}
\NewDocumentCommand \DrawSeqPath { O{} m }
  {
    \begin{scope}[#1]
      \coordinate (seqpath-current) at (\pgfkeysvalueof{/tikz/seqpath/start});
      \__seqpath_step:w #2 \q_stop
    \end{scope}
  }
\NewDocumentCommand \__seqpath_step:w { m }
  {
    \token_if_eq_meaning:NNF #1 \q_stop
      {
        \peek_meaning_ignore_spaces:NTF \c_group_begin_token
          {\__seqpath_edge:nn {#1}}
          {\__seqpath_edge:nn {#1}{1}}
      }
  }
\NewDocumentCommand \__seqpath_edge:nn { m m O{} }
  {
    \str_case:nnF {#1}
      {
        {H}{\path ([shift={(#2,0)}]seqpath-current) coordinate (seqpath-next);}
        {U}{\path ([shift={(#2,0)}]seqpath-current~|-~0,1) coordinate (seqpath-next);}
        {D}{\path ([shift={(#2,0)}]seqpath-current~|-~0,-1) coordinate (seqpath-next);}
      }
      {\msg_fatal:nnn {seqpath} {step} {#1}}
    \path (seqpath-current) edge[draw,#3] (seqpath-next);
    \coordinate (seqpath-current) at (seqpath-next);
    \__seqpath_step:w
  }
\ExplSyntaxOff
