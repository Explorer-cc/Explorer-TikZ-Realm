\documentclass[tikz,border=5pt]{standalone}
\usetikzlibrary{arrows.meta,bending}
\begin{document}

\def\NN{5}
\def\RR{5}
\def\rr{0.5}
\def\myshort{\dimexpr\rr cm+0.5\pgflinewidth}%
\def\deltaang{\fpeval{360/\NN}}
\begin{tikzpicture}[
    thick,>=Stealth,shortstyle/.style={shorten >=\myshort,shorten <=\myshort}]
    \foreach \i in {1,...,\NN}{
        \def\ang{\fpeval{\deltaang*(\i-1)}}
        \path[draw,fill=gray!75] (\ang:\RR cm) circle[radius=\rr cm];
        \draw[<->,shortstyle,violet] (\ang:\RR cm) arc[radius=\RR cm,start angle=\ang,delta angle=\deltaang];
        \foreach[parse=true] \j in {2,...,{\NN-2}}{
            \def\nextang{\fpeval{\ang+\deltaang*\j}}
            \draw[shortstyle,<->] (\ang:\RR cm) -- (\nextang:\RR cm);
        }
    }
    \draw[magenta,thin] circle[radius=\RR cm];
\end{tikzpicture}

\end{document}
