% Four-point star: x, y, size, color.
\newcommand{\sparkle}[4]{%
  \begin{scope}[shift={(#1,#2)},scale=#3]
    \fill[#4] (0,1)
      .. controls (.12,.23) and (.23,.12) .. (1,0)
      .. controls (.23,-.12) and (.12,-.23) .. (0,-1)
      .. controls (-.12,-.23) and (-.23,-.12) .. (-1,0)
      .. controls (-.23,.12) and (-.12,.23) .. (0,1);
  \end{scope}%
}
