\newcommand{\PanelWidth}{65pt}
\newcommand{\CellGap}{6pt}
\newcommand{\CellPadding}{3pt}
\newcommand{\ContentPadding}{2pt}
\newlength{\CodonLineHeight}
\setlength{\CodonLineHeight}{14pt}

\newcommand{\GroupLabel}[2]{%
  \hspace{1pt}%
  \parbox[c][\dimexpr#1\CodonLineHeight\relax][c]{26pt}{%
    \raggedright\small\bfseries#2%
  }%
}

\newcommand{\ChartCell}[3][\dimexpr4\CodonLineHeight+16pt\relax]{%
  \parbox[c][\dimexpr#1+\CellGap\relax][c]{\dimexpr#2+\CellGap+\CellPadding*2+\ContentPadding*2\relax}{%
    \centering
    \parbox[c][#1][c]{\dimexpr#2+\ContentPadding*2\relax}{\centering#3}%
  }%
}
\newcommand{\CodonLines}[1]{%
  \begingroup
  \setlength{\tabcolsep}{0pt}%
  \begin{tabular}{@{}>{\rule[-.3\CodonLineHeight]{0pt}{\CodonLineHeight}}c@{}}%
    #1
  \end{tabular}%
  \endgroup
}
\newcommand{\BraceGroup}[2]{%
  \hbox{\parbox[c][\dimexpr#1\CodonLineHeight\relax][c]{10pt}{%
    \centering
    \raisebox{-2pt}[0pt][0pt]{%
      \(\nulldelimiterspace=0pt
        \smash{\left.\vcenter{\hbox{%
          \rule{0pt}{\dimexpr#1\CodonLineHeight-3pt\relax}%
        }}\right\}}%
      \)%
    }%
  }\GroupLabel{#1}{#2}}%
}
\newcommand{\BraceBlank}[2]{%
  \hbox{%
    \parbox[c][\dimexpr#1\CodonLineHeight\relax][c]{10pt}{\mbox{}}%
    \GroupLabel{#1}{#2}%
  }%
}
\newcommand{\CodonPanel}[2]{%
  \ChartCell{\PanelWidth}{%
    \CodonLines{#1}\hspace{2pt}%
    \parbox[c][\dimexpr4\CodonLineHeight\relax][c]{37pt}{%
      \offinterlineskip#2%
    }%
  }%
}
\newcommand{\RowIndex}[1]{{%
  \renewcommand{\CellPadding}{0pt}%
  \ChartCell{18pt}{#1}%
}}
\newcommand{\ColumnIndex}[1]{\ChartCell[18pt]{\PanelWidth}{#1}}
\newcommand{\ThirdIndex}[1]{\RowIndex{\CodonLines{#1}}}
