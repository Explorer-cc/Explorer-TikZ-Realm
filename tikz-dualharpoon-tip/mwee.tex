\documentclass[tikz,border=5pt]{standalone}
\usetikzlibrary{arrows.meta,bending,nfold}
\makeatletter
\pgfset{%
flip arrows/.code=\let\pgf@temp\pgf@start@tip@sequence%
\let\pgf@start@tip@sequence\pgf@end@tip@sequence%
\let\pgf@end@tip@sequence\pgf@temp%
}
% If a user does \draw[relarrow', Foo-Baz] … ; the Foo-Baz specification gets ignored anyway because the actual path doesn't get drawn.
\tikzset{
  shrink/.style={thick,shorten >=5pt,shorten <=5pt},
  offset/.code=\tikz@addoption{\pgfgetpath\tikz@temp\pgfsetpath\pgfutil@empty\pgfoffsetpath\tikz@temp{#1}},
  foo/.tip={Straight Barb[right,length=5pt]},
  @dualharpoon/.style 2 args={
    draw=none,
    postaction={path only,draw,foo-,#2,shrink,offset=+#1ex},
    postaction={path only,draw,-foo,#2,shrink,offset=-#1ex}},
  dualharpoon/.default=1, dualharpoon'/.default=1,
  dualharpoon/.style={@dualharpoon={#1}{}},
  dualharpoon'/.style={@dualharpoon={#1}{arrows={[swap]},flip arrows}},
  relarrow/.style ={dualharpoon ,orange,thick,#1},
  relarrow'/.style={dualharpoon',orange,thick,#1},
}
\makeatother
\begin{document}
\begin{tikzpicture}
\node[fill=cyan,    circle] (A) at (-2,-1) {A};
\node[fill=magenta, circle] (B) at ( 2,-1) {B};
\node[fill=cyan,    circle] (C) at (-2, 1) {C};
\node[fill=magenta, circle] (D) at ( 2, 1) {D};
\draw[relarrow=teal] (A) -- (B);
\draw[relarrow']     (C) -- (D);
\end{tikzpicture}
\end{document}
